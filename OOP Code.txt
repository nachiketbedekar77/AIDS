class Student1 { 
    String name; 
    int age; 
    int total_marks; 
    long mobile_no; 
    float percentage; 
 
    Student1() { 
        System.out.println("The Default Constructor:"); 
        name = "Unknown"; 
        age = 0; 
        total_marks = 0; 
        mobile_no = 0; 
        percentage = 0.0f; 
    } 
 
    Student1(String n, int a, int tm, long mn, float p) { 
        System.out.println("The Parameterized Constructor:"); 
        name = n; 
        age = a; 
        total_marks = tm; 
        mobile_no = mn; 
        percentage = p; 
    } 
 
    Student1(Student1 obj) { 
        System.out.println("The Copy Constructor:"); 
        name = obj.name; 
        age = obj.age; 
        total_marks = obj.total_marks; 
        mobile_no = obj.mobile_no; 
        percentage = obj.percentage; 
    } 
 
    void display() { 
        System.out.println("Student Name: " + name); 
        System.out.println("Student Age: " + age); 
        System.out.println("Total Marks: " + total_marks); 
        System.out.println("Mobile Number: " + mobile_no); 
        System.out.println("Percentage: " + percentage + "%\n"); 
    } 
}

abstract class Shape { 
  
 public abstract  double calculateArea(); 
  
 public void display() { 
  System.out.println("Calculating area for the shape"); 
 } 
} 
 
class Circle extends Shape { 
 private double radius; 
  
 public Circle(double radius) { 
   this.radius = radius; 
 
  } 
   
  @Override  
  public double calculateArea() { 
   return Math.PI * radius * radius ; 
  } 
 } 
 
class Rectangle extends Shape { 
 private double length; 
 private double breadth; 
  
 public Rectangle (double length, double breadth) { 
  this.length = length; 
  this.breadth = breadth; 
   
 } 
  
 @Override  
 public double calculateArea () { 
  return length * breadth; 
 } 
}

interface Drawable { 
void draw(); 
} 
interface Resizable { 
void resize(int percentage); 
} 
class Circle8 implements Drawable { 
private int radius; 
public Circle8 (int radius) { 
this.radius = radius; 
} 
@Override 
public void draw() { 
System.out.println("Drawing a circle with radius: " + radius); 
} 
} 
class Rectangle8 implements Drawable, Resizable { 
private double length; 
private double breadth; 
public Rectangle8 (double length, double breadth) { 
this.length = length; 
this.breadth = breadth; 
} 
  
 @Override 
 public void draw () { 
  System.out.println("Drawing a rectangle [Length: " + length + ", Breadth: " + breadth + 
"]"); 
 } 
 
    @Override 
    public void resize(int percentage) { 
        System.out.println("Resizing Rectangle by " + percentage + "%..."); 
        length = length + (length * percentage / 100.0); 
        breadth = breadth + (breadth * percentage / 100.0); 
    } 
} 
 
public class Prac8 { 
  
    public static void main(String[] args) { 
         
        System.out.println("--- Single Interface Implementation ---"); 
        Circle8 myCircle = new Circle8(5); 
        myCircle.draw(); 
         
        System.out.println("\n--- Multiple Inheritance via Interfaces ---"); 
        Rectangle8 myRect = new Rectangle8(10.0, 20.0); 
         
        myRect.draw();  
        myRect.resize(50);  
        myRect.draw();  
    }

package sybtech; 
 
class InsuAicientBalanceException extends Exception { 
    public InsuAicientBalanceException(String message) { 
        super(message); 
    } 
} 
 
class InvalidTransactionException extends Exception { 
    public InvalidTransactionException(String message) { 
        super(message); 
    } 
} 
 
class BankAccount { 
    private String accountHolder; 
    private double balance; 
 
    public BankAccount(String accountHolder, double initialBalance) { 
        this.accountHolder = accountHolder; 
        this.balance = initialBalance; 
    } 
 
    public void deposit(double amount) throws InvalidTransactionException { 
        if (amount <= 0) { 
            throw new InvalidTransactionException("Deposit amount must be greater than zero."); 
        } 
        balance += amount; 
        System.out.println("Successfully deposited: Rs." + amount); 
        System.out.println("Updated Balance: Rs." + balance); 
    } 
 
    public void withdraw(double amount) throws InvalidTransactionException, 
InsuAicientBalanceException { 
        if (amount <= 0) { 
            throw new InvalidTransactionException("Withdrawal amount must be greater than zero."); 
        } 
        if (amount > balance) { 
            throw new InsuAicientBalanceException("InsuAicient funds! Available balance: Rs." + balance); 
        } 
        balance -= amount; 
        System.out.println("Successfully withdrew: Rs." + amount); 
        System.out.println("Remaining Balance: Rs." + balance); 
    } 
} 
 
public class Prac10 { 
    public static void main(String[] args) { 
        BankAccount account = new BankAccount("Rohan Verma", 5000.0); 
 
        System.out.println("--- Transaction 1: Valid Deposit ---"); 
        try { 
            account.deposit(2000.0); 
        } catch (InvalidTransactionException e) { 
            System.out.println("Transaction Error: " + e.getMessage()); 
        } finally { 
            System.out.println("Transaction 1 process completed.\n"); 
        } 
 
        System.out.println("--- Transaction 2: InsuAicient Balance ---"); 
        try { 
            account.withdraw(10000.0); 
        } catch (InsuAicientBalanceException | InvalidTransactionException e) { 
            System.out.println("Transaction Error: " + e.getMessage()); 
        } finally { 
            System.out.println("Transaction 2 process completed.\n"); 
        } 
 
        System.out.println("--- Transaction 3: Invalid Amount ---"); 
        try { 
            account.deposit(-500.0); 
        } catch (InvalidTransactionException e) { 
            System.out.println("Transaction Error: " + e.getMessage()); 
        } finally { 
            System.out.println("Transaction 3 process completed."); 
        } 
    } 
}
