#include <iostream>
#include <string>
using namespace std;

#define MAX 100

class UndoStack {
private:
    string actions[MAX];
    int top;

public:
    UndoStack() {
        top = -1;
    }

    void push(string action) {
        if (top >= MAX - 1) {
            cout << "Stack Overflow. Cannot store more actions.\n";
        } else {
            top++;
            actions[top] = action;
            cout << "Action Performed: " << action << "\n";
        }
    }

    void pop() {
        if (top < 0) {
            cout << "Stack Underflow. No actions to undo.\n";
        } else {
            cout << "Undo Performed: Reverted '" << actions[top] << "'\n";
            top--;
        }
    }

    void display() {
        if (top < 0) {
            cout << "No recent actions in history.\n";
        } else {
            cout << "Recent Actions (Most recent first):\n";
            for (int i = top; i >= 0; i--) {
                cout << "- " << actions[i] << "\n";
            }
        }
        cout << "\n";
    }
};

int main() {
    UndoStack editor;

    editor.push("Typed 'Hello'");
    editor.push("Typed ' '");
    editor.push("Typed 'World'");
    
    cout << "\n";
    editor.display();

    editor.pop();
    
    cout << "\n";
    editor.display();

    editor.pop();
    editor.pop();
    
    cout << "\nTrying to undo again:\n";
    editor.pop(); 

    return 0;
}

#include <iostream>
#include <stack>
#include <string>
using namespace std;

bool isBalanced(string expr) {
    stack<char> s;

    for (char ch : expr) {
        if (ch == '(' || ch == '{' || ch == '[') {
            s.push(ch);
        } 
        else if (ch == ')' || ch == '}' || ch == ']') {
            if (s.empty()) {
                return false;
            }
            char top = s.top();
            if ((ch == ')' && top == '(') ||
                (ch == '}' && top == '{') ||
                (ch == ']' && top == '[')) {
                s.pop();
            } else {
                return false;
            }
        }
    }

    return s.empty();
}

int main() {

    cout << "Enter a mathematical expression: ";
    getline(cin, expression);

    if (isBalanced(expression)) {
        cout << "The expression has BALANCED parentheses.\n";
    } else {
        cout << "The expression has UNBALANCED parentheses.\n";
    }

    return 0;
}

#include <iostream>
using namespace std;

struct Node {
    int coeff;
    int power;
    Node* next;
};

void insertTerm(Node*& head, int c, int p) {
    if (c == 0) return; 

    Node* newNode = new Node;
    newNode->coeff = c;
    newNode->power = p;
    newNode->next = nullptr;

    if (head == nullptr || p > head->power) {
        newNode->next = head;
        head = newNode;
    } 
    else if (p == head->power) {
        head->coeff += c;
        delete newNode;
    } 
    else {
        Node* temp = head;
        while (temp->next != nullptr && temp->next->power >= p) {
            if (temp->next->power == p) {
                temp->next->coeff += c;
                delete newNode;
                return;
            }
            temp = temp->next;
        }
        newNode->next = temp->next;
        temp->next = newNode;
    }
}

void createPolynomial(Node*& head) {
    int n, c, p;
    cout << "Enter the number of terms: ";
    cin >> n;
    
    for (int i = 0; i < n; i++) {
        cout << "Term " << i + 1 << " - Enter coefficient: ";
        cin >> c;
        cout << "Term " << i + 1 << " - Enter power: ";
        cin >> p;
        insertTerm(head, c, p);
    }
}

Node* addPolynomials(Node* p1, Node* p2) {
    Node* result = nullptr;
    
    while (p1 != nullptr) {
        insertTerm(result, p1->coeff, p1->power);
        p1 = p1->next;
    }
    
    while (p2 != nullptr) {
        insertTerm(result, p2->coeff, p2->power);
        p2 = p2->next;
    }
    
    return result;
}

    }

            if (temp->next->coeff > 0) {
                cout << " + "; 
            } else if (temp->next->coeff < 0) {
                cout << " "; 
            }
        }

}

    PrinterQueue() {
        head = nullptr;
    }

    void addJob(string docName, int prio) {
        PrintJob* newJob = new PrintJob();
        newJob->documentName = docName;
        newJob->priority = prio;
        newJob->next = nullptr;

        if (head == nullptr || prio > head->priority) {
            newJob->next = head;
            head = newJob;
        } else {
            PrintJob* temp = head;
            while (temp->next != nullptr && temp->next->priority >= prio) {
                temp = temp->next;
            }
            newJob->next = temp->next;
            temp->next = newJob;
        }
        cout << "Added '" << docName << "' with priority " << prio << ".\n";
    }

    void processJob() {
        if (head == nullptr) {
            cout << "No print jobs in the queue.\n";
            return;
        }

        PrintJob* temp = head;
        cout << "Printing: " << temp->documentName << " (Priority: " << temp->priority << ")\n";
        head = head->next;
        delete temp;
    }

    void displayQueue() {
        if (head == nullptr) {
            cout << "Printer queue is empty.\n";
            return;

        cout << "\n--- Pending Print Jobs ---\n";

        int count = 1;

            cout << count << ". " << temp->documentName << " (Priority: " << temp->priority << ")\n";
